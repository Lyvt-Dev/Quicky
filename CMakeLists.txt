cmake_minimum_required(VERSION 3.25)
project(Quicky VERSION 0.1.0 LANGUAGES CXX)
option(QUICKY_BUILD_APP "Build the native Qt Quick application" ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(quicky_core core/src/transfer.cpp)
target_include_directories(quicky_core PUBLIC core/include)
target_compile_features(quicky_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(quicky_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(quicky_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(QUICKY_BUILD_APP)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network Concurrent)
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/quicky-qt-version.txt" CONTENT "${Qt6Core_VERSION}\n")
  find_package(OpenSSL 3 REQUIRED)
  qt_standard_project_setup()
  qt_add_library(quicky_network STATIC
    apps/native/identity.cpp apps/native/identity.hpp
    apps/native/engine.cpp apps/native/engine.hpp)
  target_sources(quicky_network PRIVATE apps/native/mdns.cpp apps/native/mdns.hpp)
  target_link_libraries(quicky_network PUBLIC Qt6::Core Qt6::Network Qt6::Concurrent OpenSSL::Crypto)
  target_include_directories(quicky_network PUBLIC apps/native)
  target_compile_features(quicky_network PUBLIC cxx_std_20)
  qt_add_executable(Quicky apps/native/main.cpp apps/native/qrcode.cpp apps/native/qrcode.hpp
    third_party/qrcodegen/qrcodegen.cpp)
  target_include_directories(Quicky PRIVATE third_party/qrcodegen)
  qt_add_resources(Quicky "ui" PREFIX "/" FILES apps/native/Main.qml)
  target_link_libraries(Quicky PRIVATE quicky_network Qt6::Gui Qt6::Quick Qt6::QuickControls2)
  set_target_properties(Quicky PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
  if(ANDROID)
    set_target_properties(Quicky PROPERTIES
      QT_ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/packaging/android"
      QT_ANDROID_PACKAGE_NAME "dev.lyvt.quicky"
      QT_ANDROID_VERSION_NAME "${PROJECT_VERSION}"
      QT_ANDROID_VERSION_CODE 1
      QT_ANDROID_MIN_SDK_VERSION 28
      QT_ANDROID_TARGET_SDK_VERSION 35)
    if(ANDROID_EXTRA_LIBS)
      set_property(TARGET Quicky PROPERTY QT_ANDROID_EXTRA_LIBS "${ANDROID_EXTRA_LIBS}")
    endif()
  endif()
  install(TARGETS Quicky BUNDLE DESTINATION . RUNTIME DESTINATION bin)
endif()
add_executable(quicky-cli apps/cli/main.cpp)
target_link_libraries(quicky-cli PRIVATE quicky_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(quicky_tests tests/transfer_test.cpp)
  target_link_libraries(quicky_tests PRIVATE quicky_core)
  add_test(NAME transfer_state_and_files COMMAND quicky_tests)
  if(QUICKY_BUILD_APP)
    qt_add_executable(quicky_network_tests tests/network_test.cpp)
    target_link_libraries(quicky_network_tests PRIVATE quicky_network)
    add_test(NAME encrypted_network_transfer COMMAND quicky_network_tests)
    set_tests_properties(encrypted_network_transfer PROPERTIES TIMEOUT 60)
  endif()
endif()
